## Interface (locked)
```python
# cli/maintain.py
def load_policy(text: str) -> dict:
    """yaml text -> {"enabled": bool, "rung": int 0..3, "login": str, "max_actions_per_day": int}.
    enabled is True only if the yaml says exactly `true`. Empty, broken or not a mapping -> paused.
    Bad or out-of-range rung -> 0. Defaults: rung 0, login "", max_actions_per_day 20."""

def pr_facts(pr: dict) -> dict:
    """`gh pr view N --json author,body,reviews` -> {"author", "author_model", "approvals"}.
    Models come from a line "Model: <name>": the PR body for the author, each review body for its reviewer.
    approvals = [{"by", "model"}] for reviewers whose LATEST review is APPROVED, in order of first review.
    Missing fields -> {"author": None, "author_model": None, "approvals": []}."""

def render(actions: list, rung: int) -> list:
    """One line per action: "would {kind} {target}: {why}",
    plus "  (needs rung {r}; now {rung})" when the action's rung r is above the policy's.
    No actions -> ["nothing to do"]."""

def dry_run(repo: str, config: dict) -> int:
    """The I/O. Read-only. Steps:
    1. policy = load_policy(text of .crowd/policy/maintainer.yaml on the default branch,
       via `gh api repos/{repo}/contents/... -H "Accept: application/vnd.github.raw"`; missing -> "")
    2. items = watch.attention(watch.snapshot(repo), config, now)
    3. facts = {"prs": {N: pr_facts(gh pr view N ...)} for each pr-ready:N, "closed_drafts": set()}
    4. print a header line naming the repo and rung and saying it writes nothing,
       then "paused" if not enabled, else the render(decide(...), rung) lines. Return 0."""
```
